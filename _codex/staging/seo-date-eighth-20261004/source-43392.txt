function showFooterSticky({ callFunction, arg = null, labelButton = 'Próximo', textValue }) {
  let existingFooterSticky = document.querySelector('.footer-sticky');

  if (!existingFooterSticky) {
    const footerSticky = `
       
         
           
              
             
              ${labelButton}
             
           
         
       `;
    document.querySelector('footer').insertAdjacentHTML('beforebegin', footerSticky);
    existingFooterSticky = document.querySelector('.footer-sticky');
  }

  if (textValue !== undefined) {
    updateFooterValues(existingFooterSticky, textValue, labelButton);
  } else {
    const textSection = existingFooterSticky.querySelector('.text-section');
    if (textSection) {
      textSection.innerHTML = '    Selecione uma categoria!  ';
    }
  }
}

function updateFooterValues(existingFooterSticky, textValue, labelButton) {
  const textSection = existingFooterSticky.querySelector('.text-section');
  const buttonSubmit = existingFooterSticky.querySelector('#btn-submit');

  buttonSubmit.innerHTML = labelButton

  if (textValue.includes('R$')) {
    textSection.innerHTML = ` ${textValue} `;
  } else {
    const [category, modality] = textValue.split('|').map(str => str.trim());
    textSection.innerHTML = `
       ${category} ${modality ? ` ${modality} ` : ''}
       ${category} ${modality ? ` ${modality} ` : ''}`;
  }
}
